class Node:
    def __init__(self, label):
        self._label = label
        self._incoming = []
        self._outgoing = []

    @property
    def label(self):
        return self._label

    @label.setter
    def label(self, value):
        self._label = value

    @property
    def incoming(self):
        return self._incoming

    @property
    def outgoing(self):
        return self._outgoing

def query_path_forward(self, labels):

    results = []

    def search(node, label_index, path):

        if label_index == len(labels):
            results.append(path)
            return

        wanted_label = labels[label_index]

        for edge in node.outgoing:

            if edge.label == wanted_label:

                search(
                    edge.target,
                    label_index + 1,
                    path + [edge.target]
                )

    for start_node in self.nodes:
        search(start_node, 0, [start_node])

    return results
